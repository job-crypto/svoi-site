create temp table demo_s_ids on commit drop as
with ins as (
  insert into public.places (author_id, category, name, description, city, country, address, is_self, created_at)
  select u.id, s.category, s.name, s.description, u.city, u.country, s.address, true, now() - interval '4 days'
  from demo_s s join demo_u u on u.username = s.author
  where not exists (select 1 from public.places x where x.author_id = u.id and x.name = s.name)
  returning id, name)
select ins.id, s.reviews from ins join demo_s s on s.name = ins.name;

insert into public.place_reviews (place_id, author_id, rating, body, created_at)
select d.id, u.id, r.rating::smallint, r.body, now() - (interval '1 day' * (1 + r.n))
from demo_s_ids d
cross join lateral (
  select d.reviews[i][1] as username, d.reviews[i][2]::int as rating, d.reviews[i][3] as body, i as n
  from generate_subscripts(d.reviews, 1) as i) r
join demo_u u on u.username = r.username
on conflict do nothing;

select public.place_recount(id) from demo_s_ids;

select 'ok 027', (select count(*) from public.places where is_self) as self_places;
commit;
